1//! SCS archive reader for ETS2 HashFS format (CityHash v2).
2//!
3//! # File format (32-byte header + 32-byte flat entries)
4//!
5//! ```text
6//! Offset  Size  Field
7//! 0x00    4     magic       = b"SCS#" (0x53,0x43,0x53,0x23)
8//! 0x04    2     version     = u16 LE (typically 2)
9//! 0x06    2     salt        = u16 LE
10//! 0x08    4     hash_method = b"CITY" (0x43,0x49,0x54,0x59)
11//! 0x0C    4     entry_count = u32 LE
12//! 0x10    4     start_offset= u32 LE (absolute offset to data section)
13//! 0x14    12    padding     (ignored)
14//! 0x20    *     entries[]   = entry_count × 32 bytes each
15//! ─────────────────────────────────────────────────────────
16//! Entry layout (32 bytes each):
17//!   0x00  8  hash            = u64 LE (CityHash64 of file path)
18//!   0x08  8  offset          = u64 LE (absolute byte offset to data)
19//!   0x10  4  flags           = u32 LE (bit0=dir, bit1=compressed, bit2=verify, bit3=encrypted)
20//!   0x14  4  crc             = u32 LE (CRC32)
21//!   0x18  4  size            = u32 LE (uncompressed size)
22//!   0x1C  4  compressed_size = u32 LE (0 or ==size if uncompressed)
23//! ```
24//!
25//! Data entries are zlib-compressed when bit 1 of flags is set.
26//! Offsets are absolute within the .scs file.
27

38const MAGIC: [u8; 4] = [0x53, 0x43, 0x53, 0x23]; // "SCS#"
39const CITY: [u8; 4] = [0x43, 0x49, 0x54, 0x59];   // "CITY"
40
41/// Entry flags.
42#[allow(dead_code)]
43const FLAG_DIRECTORY: u32   = 1 << 0;
44const FLAG_COMPRESSED: u32  = 1 << 1;
45#[allow(dead_code)]
46const FLAG_VERIFY: u32      = 1 << 2;
47#[allow(dead_code)]
48const FLAG_ENCRYPTED: u32   = 1 << 3;
49
50/// One directory entry in the SCS archive (32 bytes on disk).
51#[derive(Debug, Clone)]
52struct DirEntry {
53    /// CityHash64 of the full file path.
54    #[allow(dead_code)]
55    hash: u64,
56    /// Absolute byte offset within the .scs file to the file data.
57    offset: u64,
58    /// Bit flags (directory, compressed, etc.).
59    flags: u32,
60    /// CRC32 checksum.
61    #[allow(dead_code)]
62    crc: u32,
63    /// Uncompressed file size.
64    size: u32,
65    /// Compressed size on disk (0 or == size if uncompressed).
66    compressed_size: u32,
67}
68
69/// Parsed SCS header.
70///
71/// Note: the `salt` field at byte offset 0x06 is read from the archive but
72/// not retained — it is only required during entry-hash mixing in older
73/// SCS variants and is not used by the v2 CityHash-based layout we target.
74#[derive(Debug)]
75struct Header {
76    entry_count: u32,
77    entry_table_start: u64,
78    metadata_table_start: u64,
79    entry_table_length: u32,
80    metadata_table_length: u32,
81}

83/// An opened SCS archive with an in-memory file directory (keyed by CityHash64).
84pub struct ScsArchive {
85    file: BufReader<File>,
86    by_hash: HashMap<u64, DirEntry>,
87    num_entries: u32,
88    known_path_hashes: HashMap<String, u64>,
89}
90
91impl ScsArchive {
92    /// Open and index an SCS file.
93    pub fn open(path: &Path) -> Result<Self, String> {
94        let mut file = BufReader::new(
95            File::open(path).map_err(|e| format!("cannot open {}: {}", path.display(), e))?
96        );
97
98        let header = parse_header(&mut file)?;
99
100        // 1. Read + decompress Entry table (handle zero-length for minimal test files)
101        let entry_data: Vec<u8> = if header.entry_table_length == 0 {
102            vec![]
103        } else {
104            file.seek(SeekFrom::Start(header.entry_table_start))
105                .map_err(|e| format!("seek to entry_table_start: {e}"))?;
106            let mut entry_buf = vec![0u8; header.entry_table_length as usize];
107            file.read_exact(&mut entry_buf)
108                .map_err(|e| format!("read entry_table: {e}"))?;
109            inflate::decompress_to_vec_zlib(&entry_buf)
110                .map_err(|e| format!("decompress entry_table: {e:?}"))?
111        };
112
113        // 2. Read + decompress Metadata table (handle zero-length)
114        let meta_data: Vec<u8> = if header.metadata_table_length == 0 {
115            vec![]
116        } else {
117            file.seek(SeekFrom::Start(header.metadata_table_start))
118                .map_err(|e| format!("seek to metadata_table_start: {e}"))?;
119            let mut meta_buf = vec![0u8; header.metadata_table_length as usize];
120            file.read_exact(&mut meta_buf)
121                .map_err(|e| format!("read metadata_table: {e}"))?;
122            inflate::decompress_to_vec_zlib(&meta_buf)
123                .map_err(|e| format!("decompress metadata_table: {e:?}"))?
124        };
125
126        // TEMP DEBUG: print first 32 bytes of meta_data to understand layout
127        if !meta_data.is_empty() {
128            eprintln!("META first 32: {:02X?}", &meta_data[..meta_data.len().min(32)]);
129        }
130
131        // 3. Parse entries (16 B each) and resolve MainMetadata
132        let mut by_hash = HashMap::new();
133        let entry_count = (entry_data.len() / 16) as u32;
134        for i in 0..entry_count {
135            let base = (i as usize) * 16;
136            if base + 16 > entry_data.len() {
137                break;
138            }
139            let hash = u64::from_le_bytes(entry_data[base..base + 8].try_into().unwrap());
140            let metadata_index = u32::from_le_bytes(entry_data[base + 8..base + 12].try_into().unwrap());
141            let metadata_count = u16::from_le_bytes(entry_data[base + 12..base + 14].try_into().unwrap());
142            let flags = u16::from_le_bytes(entry_data[base + 14..base + 16].try_into().unwrap());
143
144            // Resolve first MainMetadata chunk (metadata_index * 4)
145            if let Some(dir_entry) = resolve_main_metadata(&meta_data, metadata_index, metadata_count, flags) {
146                if dir_entry.size > 0 {
147                    by_hash.insert(hash, dir_entry);
148                }
149            }
150        }
151
152        Ok(ScsArchive {
153            file,
154            by_hash,
155            num_entries: header.entry_count,
156            known_path_hashes: HashMap::new(),
157        })
158    }
159
160    /// Open an SCS file, returning the new Error type.
161    pub fn open_err(path: &Path) -> ParserResult<Self> {
162        Self::open(path).map_err(Error::ArchiveFormat)
163    }
164
165    /// Number of entries in the archive.
166    pub fn num_entries(&self) -> u32 { self.num_entries }
167
168    /// Return all hashes present in the archive directory.
169    pub fn entry_hashes(&self) -> Vec<u64> {
170        self.by_hash.keys().copied().collect()
171    }
172
173    /// Register a known path→hash mapping (populated by probing).
174    fn register_path(&mut self, path: &str) {
175        let hash = cityhash64(path.as_bytes());
176        if self.by_hash.contains_key(&hash) {
177            self.known_path_hashes.insert(path.to_string(), hash);
178        }
179    }
180
181    /// Read a file from the archive by its CityHash64 hash directly.
182    pub fn read_entry(&mut self, hash: u64) -> Result<Vec<u8>, String> {
183        let entry = self
184            .by_hash
185            .get(&hash)
186            .ok_or_else(|| format!("hash 0x{hash:016X} not found in archive"))?;
187
188        let abs_offset = entry.offset;
189        self.file
190            .seek(SeekFrom::Start(abs_offset))
191            .map_err(|e| format!("seek to {abs_offset}: {e}"))?;
192
193        let is_compressed = (entry.flags & FLAG_COMPRESSED) != 0;
194        let read_len = if is_compressed {
195            entry.compressed_size as usize
196        } else {
197            entry.size as usize
198        };
199
200        if read_len == 0 {
201            return Ok(Vec::new());
202        }
203
204        let mut buf = vec![0u8; read_len];
205        self.file
206            .read_exact(&mut buf)
207            .map_err(|e| format!("read: {e}"))?;
208
209        if !is_compressed || entry.compressed_size == 0 || entry.compressed_size == entry.size {
210            return Ok(buf);
211        }
212
213        inflate::decompress_to_vec_zlib(&buf)
214            .map_err(|e| format!("deflate: {e:?}"))
215    }
216
217    /// Read a file from the archive by its full path.
218    ///
219    /// The path is hashed with CityHash64 and matched against the directory.
220    pub fn read_file(&mut self, filename: &str) -> Result<Vec<u8>, String> {
221        let hash = cityhash64(filename.as_bytes());
222        self.read_entry(hash)
223            .map_err(|e| format!("{filename}: {e}"))
224    }
225
226    /// List common ETS2 file paths for probing.
227    pub fn list_known_files(&self) -> Vec<String> {
228        vec![
229            "def/world/road.sii".into(),
230            "def/world/prefab.sii".into(),
231            "def/world/sign.sii".into(),
232            "def/world/semaphore_profile.sii".into(),
233        ]
234    }
235}

241impl ArchiveReader for ScsArchive {
242    fn read_file(&mut self, logical_path: &str) -> ParserResult<Vec<u8>> {
243        self.register_path(logical_path);
244        let hash = cityhash64(logical_path.as_bytes());
245        let entry = self
246            .by_hash
247            .get(&hash)
248            .ok_or_else(|| Error::FileNotFound(logical_path.into()))?;
249
250        let abs_offset = entry.offset;
251        self.file
252            .seek(SeekFrom::Start(abs_offset))
253            .map_err(Error::Io)?;
254
255        let is_compressed = (entry.flags & FLAG_COMPRESSED) != 0;
256        let read_len = if is_compressed {
257            entry.compressed_size as usize
258        } else {
259            entry.size as usize
260        };
261
262        if read_len == 0 {
263            return Ok(Vec::new());
264        }
265
266        let mut buf = vec![0u8; read_len];
267        self.file.read_exact(&mut buf).map_err(Error::Io)?;
268
269        if !is_compressed || entry.compressed_size == 0 || entry.compressed_size == entry.size {
270            return Ok(buf);
271        }
272
273        inflate::decompress_to_vec_zlib(&buf).map_err(|e| Error::Decompression(format!("{e:?}")))
274    }
275
276    fn find_files_starting_with(&self, prefix: &str) -> Vec<String> {
277        let mut result: Vec<String> = self
278            .known_path_hashes
279            .keys()
280            .filter(|p| p.starts_with(prefix))
281            .cloned()
282            .collect();
283        result.sort();
284        result
285    }
286
287    fn entry_count(&self) -> usize {
288        self.num_entries as usize
289    }
290
291    fn read_file_with_texture(&mut self, logical_path: &str) -> ParserResult<(Vec<u8>, Option<Vec<u8>>)> {
292        let data = self.read_file(logical_path)?;
293        Ok((data, None))
294    }
295}

301fn parse_header(r: &mut BufReader<File>) -> Result<Header, String> {
302    let mut magic = [0u8; 4];
303    r.read_exact(&mut magic).map_err(|e| format!("io: {e}"))?;
304    if magic != MAGIC {
305        return Err(format!("invalid SCS magic: {magic:02X?}"));
306    }
307
308    let version = read_u16(r)?;
309    if version != 2 {
310        return Err(format!("unsupported SCS version: {version} (expected 2)"));
311    }
312
313    // Skip 2-byte salt field at offset 0x06 (not used in the v2 CityHash layout).
314    let _salt = read_u16(r)?;
315
316    let mut hm = [0u8; 4];
317    r.read_exact(&mut hm).map_err(|e| format!("io: {e}"))?;
318    if hm != CITY {
319        return Err(format!("unsupported hash method: {hm:02X?} (expected CITY)"));
320    }
321
322    let entry_count = read_u32(r)?;
323    let entry_table_length = read_u32(r)?;
324    let metadata_table_length = read_u32(r)?;
325
326    // Table starts are u32 in the v2 header (not u64)
327    let entry_table_start = read_u32(r)? as u64;
328    let metadata_table_start = read_u32(r)? as u64;
329
330    Ok(Header {
331        entry_count,
332        entry_table_start,
333        metadata_table_start,
334        entry_table_length,
335        metadata_table_length,
336    })
337}

357/// Resolve MainMetadata for an entry and return a DirEntry with correct absolute offset.
358/// Layout (assumed 12+ bytes starting at metadata_index*4):
359///   [0..4] compressed_size (low 28 bits + flags in high 4)
360///   [4..8] uncompressed_size (low 28 bits + flags)
361///   [8..12] offset_block (data offset = offset_block * 16)
362fn resolve_main_metadata(
363    meta_data: &[u8],
364    metadata_index: u32,
365    metadata_count: u16,
366    entry_flags: u16,
367) -> Option<DirEntry> {
368    if metadata_count == 0 {
369        return None;
370    }
371    let base = (metadata_index as usize) * 4;
372    if base + 12 > meta_data.len() {
373        return None;
374    }
375
376    let compressed_raw = u32::from_le_bytes([
377        meta_data[base],
378        meta_data[base + 1],
379        meta_data[base + 2],
380        meta_data[base + 3],
381    ]);
382    let uncompressed_raw = u32::from_le_bytes([
383        meta_data[base + 4],
384        meta_data[base + 5],
385        meta_data[base + 6],
386        meta_data[base + 7],
387    ]);
388    let offset_block = u32::from_le_bytes([
389        meta_data[base + 8],
390        meta_data[base + 9],
391        meta_data[base + 10],
392        meta_data[base + 11],
393    ]);
394
395    let is_compressed = (compressed_raw & 0x10000000) != 0; // flag bit
396    let compressed_size = compressed_raw & 0x0FFFFFFF;
397    let uncompressed_size = uncompressed_raw & 0x0FFFFFFF;
398
399    let mut abs_offset = (offset_block as u64) * 16;
400    // Try alternative positions if the first guess is invalid
401    if abs_offset > 300_000_000 || abs_offset == 0 {
402        // try offset at base+4 as u32
403        if base + 8 <= meta_data.len() {
404            let alt = u32::from_le_bytes([meta_data[base+4], meta_data[base+5], meta_data[base+6], meta_data[base+7]]);
405            let alt_off = (alt as u64) * 16;
406            if alt_off > 0 && alt_off < 300_000_000 {
407                abs_offset = alt_off;
408            }
409        }
410    }
411    if abs_offset > 300_000_000 || abs_offset == 0 {
412        return None;
413    }
414
415    let size = if is_compressed { compressed_size } else { uncompressed_size };
416    let compressed_size_field = if is_compressed { compressed_size } else { 0 };
417
418    Some(DirEntry {
419        hash: 0,
420        offset: abs_offset,
421        flags: entry_flags as u32,
422        crc: 0,
423        size,
424        compressed_size: compressed_size_field,
425    })
426}

432const K0: u64 = 0xc3a5c85c97cb3127;
433const K1: u64 = 0xb492b66fbe98f273;
434const K2: u64 = 0x9ae16a3b2f90404f;
435
436/// CityHash64 — the hash function used by the ETS2 SCS HashFS directory.
437pub fn cityhash64(data: &[u8]) -> u64 {
438    let len = data.len();
439    if len <= 16 { hash_0_16(data) }
440    else if len <= 32 { hash_17_32(data) }
441    else if len <= 64 { hash_33_64(data) }
442    else { hash_above_64(data) }
443}
444
445fn hash_0_16(s: &[u8]) -> u64 {
446    let len = s.len();
447    if len >= 8 {
448        let mul = K2.wrapping_add((len as u64).wrapping_mul(2));
449        let a = u64le_slice(s, 0).wrapping_add(K2);
450        let b = u64le_slice(s, len - 8);
451        let c = b.rotate_right(37).wrapping_mul(mul).wrapping_add(a);
452        let d = (a.rotate_right(25).wrapping_add(b)).wrapping_mul(mul);
453        return hash128to64(c, d);
454    }
455    if len >= 4 {
456        let mul = K2.wrapping_add((len as u64).wrapping_mul(2));
457        let a = u32le_slice(s, 0) as u64;
458        return hash128to64((len as u64).wrapping_add(a << 3), u32le_slice(s, len - 4) as u64).wrapping_mul(mul);
459    }
460    if len > 0 {
461        let a = s[0] as u64;
462        let b = s[len >> 1] as u64;
463        let c = s[len - 1] as u64;
464        let y = a.wrapping_add(b << 8);
465        let z = (len as u64).wrapping_add(c << 2);
466        return shift_mix(y.wrapping_mul(K2) ^ z.wrapping_mul(K0)).wrapping_mul(K0);
467    }
468    K2
469}

471fn hash_17_32(s: &[u8]) -> u64 {
472    let len = s.len();
473    let mul = K2.wrapping_add((len as u64).wrapping_mul(2));
474    let a = u64le_slice(s, 0).wrapping_mul(K1);
475    let b = u64le_slice(s, 8);
476    let c = u64le_slice(s, len - 8).wrapping_mul(mul);
477    let d = u64le_slice(s, len - 16).wrapping_mul(K2);
478    hash128to64(
479        (a ^ b).rotate_right(43).wrapping_add(c ^ d).wrapping_add(d.wrapping_mul(mul)),
480        a.wrapping_add(b.rotate_right(18)).wrapping_add(c).wrapping_mul(mul),
481    )
482}
483
484fn hash_33_64(s: &[u8]) -> u64 {
485    let len = s.len();
486    let mul = K2.wrapping_add((len as u64).wrapping_mul(2));
487    let a = u64le_slice(s, 0).wrapping_mul(K2);
488    let b = u64le_slice(s, 8);
489    let c = u64le_slice(s, len - 24);
490    let d = u64le_slice(s, len - 32);
491    let e = u64le_slice(s, 16).wrapping_mul(K2);
492    let f = u64le_slice(s, 24).wrapping_mul(9);
493    let g = u64le_slice(s, len - 8);
494    let h = u64le_slice(s, len - 16).wrapping_mul(mul);
495    let u = shift_mix(a.wrapping_add(g).wrapping_add(e.wrapping_mul(b))).wrapping_mul(mul).wrapping_add(d);
496    let v = shift_mix(b.wrapping_add(h).wrapping_add(f.wrapping_mul(c))).wrapping_mul(mul).wrapping_add(a);
497    let mut result = u ^ v;
498    result = result.wrapping_mul(mul).wrapping_add(h);
499    result = shift_mix(result).wrapping_mul(mul).wrapping_add(d);
500    result
501}

503fn hash_above_64(s: &[u8]) -> u64 {
504    let len = s.len();
505    let mul = K2.wrapping_add((len as u64).wrapping_mul(2));
506    let mut a = u64le_slice(s, 0).wrapping_mul(K2);
507    let mut b = u64le_slice(s, 8);
508    let mut c = u64le_slice(s, len - 24);
509    let mut d = u64le_slice(s, len - 32);
510    let mut e = u64le_slice(s, 16).wrapping_mul(K2);
511    let mut f = u64le_slice(s, 24).wrapping_mul(9);
512    let mut g = u64le_slice(s, len - 8);
513    let mut h = u64le_slice(s, len - 16).wrapping_mul(mul);
514
515    let end = len - 64;
516    let mut offset = 32;
517    while offset <= end {
518        a = a.wrapping_add(u64le_slice(s, offset).wrapping_mul(K0)).rotate_right(33).wrapping_mul(K1);
519        b = b.wrapping_add(u64le_slice(s, offset + 8).wrapping_mul(K1)).rotate_right(33).wrapping_mul(K2);
520        c = c.wrapping_add(u64le_slice(s, offset + 16).wrapping_mul(K2)).rotate_right(33).wrapping_mul(K0);
521        d = d.wrapping_add(u64le_slice(s, offset + 24).wrapping_mul(K0)).rotate_right(33);
522        e = e.wrapping_add(u64le_slice(s, offset + 32).wrapping_mul(K0)).rotate_right(33).wrapping_mul(K1);
523        f = f.wrapping_add(u64le_slice(s, offset + 40).wrapping_mul(K1)).rotate_right(33).wrapping_mul(K2);
524        g = g.wrapping_add(u64le_slice(s, offset + 48).wrapping_mul(K2)).rotate_right(33).wrapping_mul(K0);
525        h = h.wrapping_add(u64le_slice(s, offset + 56).wrapping_mul(K0)).rotate_right(33);
526        offset += 64;
527    }
528
529    let u = shift_mix(a.wrapping_add(g).wrapping_add(e.wrapping_mul(b))).wrapping_mul(mul).wrapping_add(d);
530    let v = shift_mix(b.wrapping_add(h).wrapping_add(f.wrapping_mul(c))).wrapping_mul(mul).wrapping_add(a);
531    let mut result = u ^ v;
532    result = result.wrapping_mul(mul).wrapping_add(h);
533    result = shift_mix(result).wrapping_mul(mul).wrapping_add(d);
534    result
535}

537fn hash128to64(lo: u64, hi: u64) -> u64 {
538    const K3: u64 = 0x9ddfea08eb382d69;
539    let mut a = (lo ^ hi).wrapping_mul(K3);
540    a ^= a >> 47;
541    let mut b = (hi ^ a).wrapping_mul(K3);
542    b ^= b >> 47;
543    b.wrapping_mul(K3)
544}

546fn shift_mix(v: u64) -> u64 { v ^ (v >> 47) }

557// Sector paths
558// ---------------------------------------------------------------------------

560/// Return all candidate sector logical paths the parser knows about.
561///
562/// Iterates the regular ETS2 sector grid (`map/europe/sec±XXXX±YYYY.data` /
563/// `.base`) and produces a list ready to be probed against an opened
564/// `ScsArchive`.
565pub fn list_map_sector_paths() -> Vec<String> {
566    let mut paths = Vec::new();
567    for x in -25..=25i32 {
568        for z in -25..=25i32 {
569            let sx = if x >= 0 { format!("+{x:04}") } else { format!("-{:04}", x.abs()) };
570            let sz = if z >= 0 { format!("+{z:04}") } else { format!("-{:04}", z.abs()) };
571            paths.push(format!("map/europe/sec{sx}{sz}.data"));
572            paths.push(format!("map/europe/sec{sx}{sz}.base"));
573            paths.push(format!("map/europe/sec{sx}{sz}.aux"));
574            paths.push(format!("map/europe/sec{sx}{sz}.desc"));
575        }
576    }
577    paths.extend(vec![
578        "def/map.sii".into(), "def/world/road.sii".into(),
579        "def/world/prefab.sii".into(), "def/world/sign.sii".into(),
580    ]);
581    paths
582}
