1//! Graph construction from parsed ETS2 map data.
2//!
3//! Converts raw `MapData` (nodes, roads, prefabs) into the topological
4//! `GraphData` representation used by the autopilot.
5
6use std::collections::{BTreeMap, BTreeSet, HashMap, HashSet};
7use std::hash::Hasher;
8
9use siphasher::sip::SipHasher24;

11use crate::graph_schema::{GraphData, GraphEdge, GraphMetrics, GraphNode, QualityMeta, QualityReport};
12// MapPrefab, MapRoad are used in test constructors below.
13#[allow(unused_imports)]
14use crate::json_export::{MapData, MapNode, MapPrefab, MapRoad};

16/// Fixed keys for deterministic SipHash-based edge UID generation.
17const SIP_KEY_A: u64 = 0x1234_5678;
18const SIP_KEY_B: u64 = 0x9ABC_DEF0;
19
20/// Build a graph from raw map data.
21///
22/// # Algorithm
23/// 1. Convert all `MapNode` → `GraphNode`.
24/// 2. For each road, walk consecutive node pairs and emit directed edges
25///    based on forward/backward lane counts.
26/// 3. For each prefab with >1 nodes, connect every unordered pair in both
27///    directions (unless a parallel edge already exists).
28/// 4. Validate referential integrity (no dangling node/road references).
29/// 5. Sort nodes and edges for deterministic output.
30/// 6. Generate `edge_uid` deterministically via SipHash.
31pub fn build_graph(map: &MapData) -> Result<GraphData, String> {
32    let node_lookup: BTreeMap<u64, &MapNode> = map.nodes.iter().map(|n| (n.uid, n)).collect();
33    let road_set: BTreeSet<&str> = map.roads.iter().map(|r| r.uid.as_str()).collect();
34
35    // 1) Determine lane capacity per base node.
36    let mut node_lane_capacity: BTreeMap<u64, u32> = node_lookup.keys().map(|&uid| (uid, 1_u32)).collect();
37    for road in &map.roads {
38        for &nuid in &road.nodes {
39            if !node_lookup.contains_key(&nuid) {
40                return Err(format!("dangling node_uid {} in road {}", nuid, road.uid));
41            }
42            let road_lane_max = road
43                .lane_count_forward
44                .max(road.lane_count_backward)
45                .max(1);
46            let cap = node_lane_capacity.entry(nuid).or_insert(1);
47            *cap = (*cap).max(road_lane_max);
48        }
49    }
50
51    // 2) Build lane sub-nodes.
52    let mut used_uids: BTreeSet<u64> = node_lookup.keys().copied().collect();
53    let mut lane_node_uid: BTreeMap<(u64, u32), u64> = BTreeMap::new();
54    let mut lane_node_str: BTreeMap<(u64, u32), String> = BTreeMap::new();
55    let mut nodes: Vec<GraphNode> = Vec::new();
56
57    for (&base_uid, base_node) in &node_lookup {
58        let lane_cap = *node_lane_capacity.get(&base_uid).unwrap_or(&1);
59        for lane in 0..lane_cap {
60            let lane_uid_num = if lane == 0 {
61                base_uid
62            } else {
63                unique_lane_uid(base_uid, lane, &mut used_uids)
64            };
65            let lane_uid_text = format!("0x{:X}_lane_{}", base_uid, lane);
66
67            lane_node_uid.insert((base_uid, lane), lane_uid_num);
68            lane_node_str.insert((base_uid, lane), lane_uid_text.clone());
69            nodes.push(GraphNode {
70                uid: lane_uid_num,
71                lane_uid: Some(lane_uid_text),
72                base_uid: Some(base_uid),
73                lane_index: Some(lane),
74                x: base_node.x,
75                y: base_node.y,
76                z: base_node.z,
77            });
78        }
79    }
80
81    // 3) Build lane edges from roads.
82    let mut edges: Vec<GraphEdge> = Vec::new();
83    for road in &map.roads {
84        let road_uid = Some(road.uid.clone());
85        for w in road.nodes.windows(2) {
86            let from_base = w[0];
87            let to_base = w[1];
88            let (from_node, to_node) = match (node_lookup.get(&from_base), node_lookup.get(&to_base)) {
89                (Some(f), Some(t)) => (*f, *t),
90                _ => continue,
91            };
92            let distance_m = euclidean_3d(from_node, to_node);
93
94            let fwd = road.lane_count_forward > 0;
95            let bwd = road.lane_count_backward > 0;
96
97            if fwd {
98                for lane in 0..road.lane_count_forward {
99                    let from_uid = *lane_node_uid.get(&(from_base, lane)).ok_or_else(|| {
100                        format!("missing lane node for {} lane {}", from_base, lane)
101                    })?;
102                    let to_uid = *lane_node_uid.get(&(to_base, lane)).ok_or_else(|| {
103                        format!("missing lane node for {} lane {}", to_base, lane)
104                    })?;
105                    edges.push(make_edge(EdgeParams {
106                        from_uid,
107                        to_uid,
108                        from_lane_uid: lane_node_str.get(&(from_base, lane)).cloned(),
109                        to_lane_uid: lane_node_str.get(&(to_base, lane)).cloned(),
110                        road_uid: road_uid.clone(),
111                        distance_m,
112                        direction: "forward",
113                        lane_count: road.lane_count_forward,
114                        speed_limit_kmh: road.speed_limit,
115                        lane: Some(lane),
116                        from_lane: None,
117                        to_lane: None,
118                        flags: vec![format!("lane={}", lane)],
119                    }));
120                }
121            }
122
123            if bwd {
124                for lane in 0..road.lane_count_backward {
125                    let from_uid = *lane_node_uid.get(&(to_base, lane)).ok_or_else(|| {
126                        format!("missing lane node for {} lane {}", to_base, lane)
127                    })?;
128                    let to_uid = *lane_node_uid.get(&(from_base, lane)).ok_or_else(|| {
129                        format!("missing lane node for {} lane {}", from_base, lane)
130                    })?;
131                    edges.push(make_edge(EdgeParams {
132                        from_uid,
133                        to_uid,
134                        from_lane_uid: lane_node_str.get(&(to_base, lane)).cloned(),
135                        to_lane_uid: lane_node_str.get(&(from_base, lane)).cloned(),
136                        road_uid: road_uid.clone(),
137                        distance_m,
138                        direction: "backward",
139                        lane_count: road.lane_count_backward,
140                        speed_limit_kmh: road.speed_limit,
141                        lane: Some(lane),
142                        from_lane: None,
143                        to_lane: None,
144                        flags: vec![format!("lane={}", lane)],
145                    }));
146                }
147            }
148
149            if !fwd && !bwd {
150                let from_uid = *lane_node_uid.get(&(from_base, 0)).ok_or_else(|| {
151                    format!("missing lane node for {} lane 0", from_base)
152                })?;
153                let to_uid = *lane_node_uid.get(&(to_base, 0)).ok_or_else(|| {
154                    format!("missing lane node for {} lane 0", to_base)
155                })?;
156                edges.push(make_edge(EdgeParams {
157                    from_uid,
158                    to_uid,
159                    from_lane_uid: lane_node_str.get(&(from_base, 0)).cloned(),
160                    to_lane_uid: lane_node_str.get(&(to_base, 0)).cloned(),
161                    road_uid: road_uid.clone(),
162                    distance_m,
163                    direction: "bidirectional_unknown",
164                    lane_count: 1,
165                    speed_limit_kmh: road.speed_limit,
166                    lane: Some(0),
167                    from_lane: None,
168                    to_lane: None,
169                    flags: vec!["no_lanes_unknown".to_string(), "lane=0".to_string()],
170                }));
171                edges.push(make_edge(EdgeParams {
172                    from_uid: to_uid,
173                    to_uid: from_uid,
174                    from_lane_uid: lane_node_str.get(&(to_base, 0)).cloned(),
175                    to_lane_uid: lane_node_str.get(&(from_base, 0)).cloned(),
176                    road_uid: road_uid.clone(),
177                    distance_m,
178                    direction: "bidirectional_unknown",
179                    lane_count: 1,
180                    speed_limit_kmh: road.speed_limit,
181                    lane: Some(0),
182                    from_lane: None,
183                    to_lane: None,
184                    flags: vec!["no_lanes_unknown".to_string(), "lane=0".to_string()],
185                }));
186            }
187        }
188    }
189
190    // 4) Lane-change edges between adjacent lanes at same base node.
191    for (&base_uid, &lane_cap) in &node_lane_capacity {
192        if lane_cap <= 1 {
193            continue;
194        }
195        for lane in 0..(lane_cap - 1) {
196            let from_uid = *lane_node_uid.get(&(base_uid, lane)).ok_or_else(|| {
197                format!("missing lane node for {} lane {}", base_uid, lane)
198            })?;
199            let to_uid = *lane_node_uid.get(&(base_uid, lane + 1)).ok_or_else(|| {
200                format!("missing lane node for {} lane {}", base_uid, lane + 1)
201            })?;
202            edges.push(make_edge(EdgeParams {
203                from_uid,
204                to_uid,
205                from_lane_uid: lane_node_str.get(&(base_uid, lane)).cloned(),
206                to_lane_uid: lane_node_str.get(&(base_uid, lane + 1)).cloned(),
207                road_uid: None,
208                distance_m: 1.0,
209                direction: "lane_change",
210                lane_count: 1,
211                speed_limit_kmh: None,
212                lane: None,
213                from_lane: Some(lane),
214                to_lane: Some(lane + 1),
215                flags: vec![
216                    "lane_change".to_string(),
217                    format!("from_lane={}", lane),
218                    format!("to_lane={}", lane + 1),
219                ],
220            }));
221
222            edges.push(make_edge(EdgeParams {
223                from_uid: to_uid,
224                to_uid: from_uid,
225                from_lane_uid: lane_node_str.get(&(base_uid, lane + 1)).cloned(),
226                to_lane_uid: lane_node_str.get(&(base_uid, lane)).cloned(),
227                road_uid: None,
228                distance_m: 1.0,
229                direction: "lane_change",
230                lane_count: 1,
231                speed_limit_kmh: None,
232                lane: None,
233                from_lane: Some(lane + 1),
234                to_lane: Some(lane),
235                flags: vec![
236                    "lane_change".to_string(),
237                    format!("from_lane={}", lane + 1),
238                    format!("to_lane={}", lane),
239                ],
240            }));
241        }
242    }
243
244    // 5. Prefab interconnections use lane_0 as connector lane.
245    // Validate all prefab nodes exist before processing.
246    let mut connected_pairs: HashSet<(u64, u64)> = edges
247        .iter()
248        .map(|e| {
249            if e.from_node_uid <= e.to_node_uid {
250                (e.from_node_uid, e.to_node_uid)
251            } else {
252                (e.to_node_uid, e.from_node_uid)
253            }
254        })
255        .collect();
256
257    for prefab in &map.prefabs {
258        for &nuid in &prefab.nodes {
259            if !node_lookup.contains_key(&nuid) {
260                return Err(format!(
261                    "dangling node_uid {} in prefab {}",
262                    nuid, prefab.uid
263                ));
264            }
265        }
266    }
267
268    for prefab in &map.prefabs {
269        if prefab.nodes.len() <= 1 {
270            continue;
271        }
272        let prefab_road: Option<String> = None;
273        for i in 0..prefab.nodes.len() {
274            for j in (i + 1)..prefab.nodes.len() {
275                let u = prefab.nodes[i];
276                let v = prefab.nodes[j];
277
278                // Nodes already validated above — expect is safe.
279                let u_node = map.nodes.iter().find(|n| n.uid == u).expect("validated prefab node");
280                let v_node = map.nodes.iter().find(|n| n.uid == v).expect("validated prefab node");
281                let dist = euclidean_3d(u_node, v_node);
282
283                let u_lane0 = *lane_node_uid.get(&(u, 0)).ok_or_else(|| format!("missing lane_0 for node {}", u))?;
284                let v_lane0 = *lane_node_uid.get(&(v, 0)).ok_or_else(|| format!("missing lane_0 for node {}", v))?;
285
286                let e1 = make_edge(EdgeParams {
287                    from_uid: u_lane0,
288                    to_uid: v_lane0,
289                    from_lane_uid: lane_node_str.get(&(u, 0)).cloned(),
290                    to_lane_uid: lane_node_str.get(&(v, 0)).cloned(),
291                    road_uid: prefab_road.clone(),
292                    distance_m: dist,
293                    direction: "prefab_interconnect",
294                    lane_count: 1,
295                    speed_limit_kmh: None,
296                    lane: Some(0),
297                    from_lane: None,
298                    to_lane: None,
299                    flags: Vec::new(),
300                });
301                let e2 = make_edge(EdgeParams {
302                    from_uid: v_lane0,
303                    to_uid: u_lane0,
304                    from_lane_uid: lane_node_str.get(&(v, 0)).cloned(),
305                    to_lane_uid: lane_node_str.get(&(u, 0)).cloned(),
306                    road_uid: prefab_road.clone(),
307                    distance_m: dist,
308                    direction: "prefab_interconnect",
309                    lane_count: 1,
310                    speed_limit_kmh: None,
311                    lane: Some(0),
312                    from_lane: None,
313                    to_lane: None,
314                    flags: Vec::new(),
315                });
316
317                // Skip if any edge between the same pair already exists (regardless of direction).
318                let pair = if u_lane0 <= v_lane0 {
319                    (u_lane0, v_lane0)
320                } else {
321                    (v_lane0, u_lane0)
322                };
323                let already_exists = connected_pairs.contains(&pair);
324                if !already_exists {
325                    edges.push(e1);
326                    edges.push(e2);
327                    connected_pairs.insert(pair);
328                }
329            }
330        }
331    }
332
333    // 6. Dangling-reference check.
334    let node_set: BTreeSet<u64> = nodes.iter().map(|n| n.uid).collect();
335    for edge in &edges {
336        if !node_set.contains(&edge.from_node_uid) {
337            return Err(format!(
338                "dangling from_node_uid {} in edge {}",
339                edge.from_node_uid, edge.edge_uid
340            ));
341        }
342        if !node_set.contains(&edge.to_node_uid) {
343            return Err(format!(
344                "dangling to_node_uid {} in edge {}",
345                edge.to_node_uid, edge.edge_uid
346            ));
347        }
348        if let Some(ref ruid) = edge.road_uid {
349            if !road_set.contains(ruid.as_str()) {
350                return Err(format!(
351                    "dangling road_uid {} in edge {}",
352                    ruid, edge.edge_uid
353                ));
354            }
355        }
356    }
357
358    // 7. Stable sort — include all key fields for determinism.
359    nodes.sort_by_key(|n| n.uid);
360    edges.sort_by(|a, b| {
361        a.from_node_uid
362            .cmp(&b.from_node_uid)
363            .then_with(|| a.to_node_uid.cmp(&b.to_node_uid))
364            .then_with(|| a.from_lane_uid.cmp(&b.from_lane_uid))
365            .then_with(|| a.to_lane_uid.cmp(&b.to_lane_uid))
366            .then_with(|| a.road_uid.cmp(&b.road_uid))
367            .then_with(|| a.direction.cmp(&b.direction))
368            .then_with(|| a.edge_uid.cmp(&b.edge_uid))
369    });
370
371    Ok(GraphData {
372        meta: QualityMeta {
373            schema_version: "1.0.0".into(),
374            map_name: "unknown".into(),
375            generated_at: String::new(),
376        },
377        nodes,
378        edges,
379    })
380}

382/// Parameters for creating a graph edge.
383struct EdgeParams<'a> {
384    from_uid: u64,
385    to_uid: u64,
386    from_lane_uid: Option<String>,
387    to_lane_uid: Option<String>,
388    road_uid: Option<String>,
389    distance_m: f64,
390    direction: &'a str,
391    lane_count: u32,
392    speed_limit_kmh: Option<f64>,
393    lane: Option<u32>,
394    from_lane: Option<u32>,
395    to_lane: Option<u32>,
396    flags: Vec<String>,
397}

399/// Helper: create a `GraphEdge` with deterministic `edge_uid`.
400fn make_edge(params: EdgeParams) -> GraphEdge {
401    let EdgeParams {
402        from_uid,
403        to_uid,
404        from_lane_uid,
405        to_lane_uid,
406        road_uid,
407        distance_m,
408        direction,
409        lane_count,
410        speed_limit_kmh,
411        lane,
412        from_lane,
413        to_lane,
414        flags,
415    } = params;
416    let edge_uid = hash_edge_uid(
417        from_uid,
418        to_uid,
419        &road_uid,
420        direction,
421        lane,
422        from_lane,
423        to_lane,
424    );
425    GraphEdge {
426        edge_uid,
427        from_node_uid: from_uid,
428        to_node_uid: to_uid,
429        from_lane_uid,
430        to_lane_uid,
431        road_uid,
432        distance_m,
433        direction: direction.to_string(),
434        lane_count,
435        speed_limit_kmh,
436        flags,
437    }
438}

440fn unique_lane_uid(base_uid: u64, lane: u32, used_uids: &mut BTreeSet<u64>) -> u64 {
441    let mut salt = 0_u64;
442    loop {
443        let mut hasher = SipHasher24::new_with_keys(SIP_KEY_A ^ 0xA5A5_5A5A, SIP_KEY_B ^ 0x5A5A_A5A5);
444        hasher.write(format!("lane|{}|{}|{}", base_uid, lane, salt).as_bytes());
445        let candidate = hasher.finish();
446        if !used_uids.contains(&candidate) {
447            used_uids.insert(candidate);
448            return candidate;
449        }
450        salt = salt.wrapping_add(1);
451    }
452}

454/// Compute a deterministic edge UID using SipHash-2-4 with fixed keys.
455///
456/// The input string is constructed as
457/// `<from>|<to>|<road>|<direction>[|<lane>][|fl:<from_lane>|tl:<to_lane>]`.
458fn hash_edge_uid(
459    from: u64,
460    to: u64,
461    road_uid: &Option<String>,
462    direction: &str,
463    lane: Option<u32>,
464    from_lane: Option<u32>,
465    to_lane: Option<u32>,
466) -> u64 {
467    let mut hasher = SipHasher24::new_with_keys(SIP_KEY_A, SIP_KEY_B);
468    hasher.write(format!("{}|{}", from, to).as_bytes());
469    if let Some(ref ruid) = road_uid {
470        hasher.write(b"|");
471        hasher.write(ruid.as_bytes());
472    } else {
473        hasher.write(b"|");
474    }
475    hasher.write(b"|");
476    hasher.write(direction.as_bytes());
477    if let Some(l) = lane {
478        hasher.write(b"|");
479        hasher.write(l.to_string().as_bytes());
480    }
481    if let Some(fl) = from_lane {
482        hasher.write(b"|fl:");
483        hasher.write(fl.to_string().as_bytes());
484    }
485    if let Some(tl) = to_lane {
486        hasher.write(b"|tl:");
487        hasher.write(tl.to_string().as_bytes());
488    }
489    hasher.finish()
490}

492fn euclidean_3d(a: &MapNode, b: &MapNode) -> f64 {
493    let dx = a.x - b.x;
494    let dy = a.y - b.y;
495    let dz = a.z - b.z;
496    (dx * dx + dy * dy + dz * dz).sqrt()
497}

499/// Write the graph to a JSON file with byte-stable output.
500pub fn write_graph_file(path: &str, graph: &GraphData) -> Result<(), String> {
501    let file = std::fs::File::create(path).map_err(|e| format!("cannot create {}: {}", path, e))?;
502    serde_json::to_writer(file, graph).map_err(|e| format!("serialization error: {}", e))
503}

515/// Compute aggregate metrics for a given graph.
516pub fn compute_graph_metrics(graph: &GraphData, build_time: std::time::Duration) -> GraphMetrics {
517    let nodes_total = graph.nodes.len();
518    let edges_total = graph.edges.len();
519
520    let density = if nodes_total > 1 {
521        edges_total as f64 / nodes_total as f64
522    } else {
523        0.0
524    };
525
526    let directed_count = graph.edges.iter().filter(|e| e.direction == "forward" || e.direction == "backward").count();
527    let unknown_count = graph.edges.iter().filter(|e| e.direction == "bidirectional_unknown").count();
528    let with_speed = graph.edges.iter().filter(|e| e.speed_limit_kmh.is_some()).count();
529
530    let pct_directed = if edges_total > 0 {
531        directed_count as f64 / edges_total as f64 * 100.0
532    } else {
533        0.0
534    };
535    let pct_unknown = if edges_total > 0 {
536        unknown_count as f64 / edges_total as f64 * 100.0
537    } else {
538        0.0
539    };
540    let pct_with_speed_limit = if edges_total > 0 {
541        with_speed as f64 / edges_total as f64 * 100.0
542    } else {
543        0.0
544    };
545
546    // Largest component ratio: simple BFS-based estimation.
547    let largest_component_ratio = largest_component_fraction(graph);
548
549    GraphMetrics {
550        nodes_total,
551        edges_total,
552        density,
553        largest_component_ratio,
554        pct_directed,
555        pct_unknown,
556        pct_with_speed_limit,
557        build_time_ms: build_time.as_secs_f64() * 1000.0,
558    }
559}

561fn largest_component_fraction(graph: &GraphData) -> f64 {
562    if graph.nodes.is_empty() {
563        return 0.0;
564    }
565
566    let mut adj: HashMap<u64, Vec<u64>> = HashMap::new();
567    for edge in &graph.edges {
568        adj.entry(edge.from_node_uid).or_default().push(edge.to_node_uid);
569        adj.entry(edge.to_node_uid).or_default().push(edge.from_node_uid);
570    }
571
572    let mut visited: HashSet<u64> = HashSet::new();
573    let mut max_size = 0usize;
574
575    for node in &graph.nodes {
576        if visited.contains(&node.uid) {
577            continue;
578        }
579        let mut stack = vec![node.uid];
580        visited.insert(node.uid);
581        let mut comp_size = 1;
582        while let Some(current) = stack.pop() {
583            if let Some(neighbors) = adj.get(&current) {
584                for &neighbor in neighbors {
585                    if visited.insert(neighbor) {
586                        stack.push(neighbor);
587                        comp_size += 1;
588                    }
589                }
590            }
591        }
592        if comp_size > max_size {
593            max_size = comp_size;
594        }
595    }
596
597    max_size as f64 / graph.nodes.len() as f64
598}

600/// Log graph metrics to stdout (only when `verbose` is true).
601pub fn log_graph_metrics(metrics: &GraphMetrics, verbose: bool) {
602    if !verbose {
603        return;
604    }
605    println!("Graph Metrics:");
606    println!("  nodes:             {}", metrics.nodes_total);
607    println!("  edges:             {}", metrics.edges_total);
608    println!("  density:           {:.3}", metrics.density);
609    println!("  largest component: {:.1}%", metrics.largest_component_ratio * 100.0);
610    println!("  directed:          {:.1}%", metrics.pct_directed);
611    println!("  unknown:           {:.1}%", metrics.pct_unknown);
612    println!("  with speed limit:  {:.1}%", metrics.pct_with_speed_limit);
613    println!("  build time:        {:.2} ms", metrics.build_time_ms);
614}

616/// Write a quality report to the given file path.
617pub fn write_quality_report(path: &str, report: &QualityReport) -> Result<(), String> {
618    let file = std::fs::File::create(path).map_err(|e| format!("cannot create {}: {}", path, e))?;
619    serde_json::to_writer(file, report).map_err(|e| format!("serialization error: {}", e))
620}

855    #[test]
856    fn test_edge_uid_deterministic() {
857        let uid1 = hash_edge_uid(1, 2, &Some("road_a".into()), "forward", None, None, None);
858        let uid2 = hash_edge_uid(1, 2, &Some("road_a".into()), "forward", None, None, None);
859        assert_eq!(uid1, uid2);
860
861        // Different direction gives different UID
862        let uid3 = hash_edge_uid(1, 2, &Some("road_a".into()), "backward", None, None, None);
863        assert_ne!(uid1, uid3);
864
865        // Different road gives different UID
866        let uid4 = hash_edge_uid(1, 2, &Some("road_b".into()), "forward", None, None, None);
867        assert_ne!(uid1, uid4);
868
869        // Different main lane gives different UIDs (normal edges)
870        let uid5 = hash_edge_uid(1, 2, &Some("road_a".into()), "forward", Some(0), None, None);
871        let uid6 = hash_edge_uid(1, 2, &Some("road_a".into()), "forward", Some(1), None, None);
872        assert_ne!(uid5, uid6);
873
874        // Different from/to lanes give different UIDs (lane-change edges)
875        let uid7 = hash_edge_uid(1, 2, &Some("road_a".into()), "lane_change", None, Some(0), Some(1));
876        let uid8 = hash_edge_uid(1, 2, &Some("road_a".into()), "lane_change", None, Some(1), Some(2));
877        assert_ne!(uid7, uid8);
878
879        // from_lane=Some(0),to_lane=None vs from_lane=None,to_lane=Some(0) → different
880        let uid9  = hash_edge_uid(1, 2, &Some("road_a".into()), "lane_change", None, Some(0), None);
881        let uid10 = hash_edge_uid(1, 2, &Some("road_a".into()), "lane_change", None, None, Some(0));
882        assert_ne!(uid9, uid10);
883    }
