1//! Catmull-Rom spline interpolation for smooth route following.
2//!
3//! Raw graph waypoints can be jagged (straight lines between nodes).
4//! A Catmull-Rom spline creates a smooth, continuous curve through the
5//! waypoints, producing natural steering angles.
6
7/// Build a dense list of (x, z) sample points along a Catmull-Rom spline
8/// passing through the given waypoints.
9///
10/// Each segment between two successive waypoints is subdivided into
11/// `subdivisions` equally-spaced samples (so `subdivisions + 1` points
12/// per segment including the start point).
13///
14/// Returns a flat Vec of (x, z) world coordinates.
15pub fn smooth_route(
16    waypoints: &[(f64, f64)],
17    subdivisions: usize,
18) -> Vec<(f64, f64)> {
19    if waypoints.len() < 2 {
20        return waypoints.to_vec();
21    }
22
23    let n = waypoints.len();
24    let mut result = Vec::with_capacity((n - 1) * (subdivisions + 1));
25
26    for i in 1..n {
27        // Catmull-Rom uses four control points: P_{i-2}, P_{i-1}, P_i, P_{i+1}.
28        // For the first segment, duplicate P0; for the last, duplicate P_{n-1}.
29        let p0 = if i >= 2 { waypoints[i - 2] } else { waypoints[0] };
30        let p1 = waypoints[i - 1];
31        let p2 = waypoints[i];
32        let p3 = if i + 1 < n { waypoints[i + 1] } else { waypoints[n - 1] };
33
34        for j in 0..=subdivisions {
35            let t = j as f64 / subdivisions as f64;
36            // Skip the first point of each segment after the first (already emitted).
37            if j == 0 && i > 1 {
38                continue;
39            }
40            let pt = catmull_rom_point(p0, p1, p2, p3, t);
41            result.push(pt);
42        }
43    }
44
45    result
46}

48/// Evaluate a single point on a Catmull-Rom spline.
49///
50/// `p0..p3` are the four control points; `t` ∈ [0, 1] is the interpolation
51/// parameter along the segment from `p1` to `p2`.
52fn catmull_rom_point(
53    p0: (f64, f64),
54    p1: (f64, f64),
55    p2: (f64, f64),
56    p3: (f64, f64),
57    t: f64,
58) -> (f64, f64) {
59    let t2 = t * t;
60    let t3 = t2 * t;
61
62    // Catmull-Rom basis matrix coefficients (uniform).
63    let cx = 0.5
64        * ((2.0 * p1.0)
65            + (-p0.0 + p2.0) * t
66            + (2.0 * p0.0 - 5.0 * p1.0 + 4.0 * p2.0 - p3.0) * t2
67            + (-p0.0 + 3.0 * p1.0 - 3.0 * p2.0 + p3.0) * t3);
68    let cz = 0.5
69        * ((2.0 * p1.1)
70            + (-p0.1 + p2.1) * t
71            + (2.0 * p0.1 - 5.0 * p1.1 + 4.0 * p2.1 - p3.1) * t2
72            + (-p0.1 + 3.0 * p1.1 - 3.0 * p2.1 + p3.1) * t3);
73
74    (cx, cz)
75}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    #[test]
86    fn test_smooth_straight_line() {
87        let wp = vec![(0.0, 0.0), (100.0, 0.0)];
88        let pts = smooth_route(&wp, 2);
89        // 2 waypoints, 1 segment, 2 subdivisions → 3 points.
90        assert_eq!(pts.len(), 3);
91        assert_eq!(pts[0], (0.0, 0.0));
92        assert_eq!(pts[2], (100.0, 0.0));
93    }
94
95    #[test]
96    fn test_smooth_single_point() {
97        let wp = vec![(5.0, 3.0)];
98        let pts = smooth_route(&wp, 4);
99        assert_eq!(pts.len(), 1);
100        assert_eq!(pts[0], (5.0, 3.0));
101    }
102
103    #[test]
104    fn test_smooth_empty() {
105        let pts = smooth_route(&[], 2);
106        assert!(pts.is_empty());
107    }
108
109    #[test]
110    fn test_catmull_rom_straight() {
111        // Four collinear points → output should be on the line.
112        let pt = catmull_rom_point((0.0, 0.0), (1.0, 0.0), (2.0, 0.0), (3.0, 0.0), 0.5);
113        assert!((pt.0 - 1.5).abs() < 0.01);
114        assert!((pt.1 - 0.0).abs() < 0.01);
115    }
116
117    #[test]
118    fn test_catmull_rom_endpoints() {
119        // t=0 should match p1, t=1 should match p2.
120        let p0 = (-1.0, -1.0);
121        let p1 = (0.0, 0.0);
122        let p2 = (2.0, 2.0);
123        let p3 = (3.0, 3.0);
124
125        let pt0 = catmull_rom_point(p0, p1, p2, p3, 0.0);
126        assert!((pt0.0 - p1.0).abs() < 0.0001, "t=0 should be p1");
127        assert!((pt0.1 - p1.1).abs() < 0.0001, "t=0 should be p1");
128
129        let pt1 = catmull_rom_point(p0, p1, p2, p3, 1.0);
130        assert!((pt1.0 - p2.0).abs() < 0.0001, "t=1 should be p2");
131        assert!((pt1.1 - p2.1).abs() < 0.0001, "t=1 should be p2");
132    }
133}
