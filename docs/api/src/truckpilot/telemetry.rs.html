1//! Telemetry data fetching from the ETS2 Funbit telemetry server.
2//!
3//! Parses the REST API JSON provided by the ETS2 Telemetry Server
4//! (default: `http://localhost:25555/api/ets2/telemetry`).
5
6use serde::Deserialize;
7
8/// Telemetry snapshot as returned by the ETS2 telemetry REST API.
9#[derive(Debug, Clone, Deserialize)]
10pub struct TelemetryData {
11    /// Truck position and orientation.
12    #[serde(rename = "truckPlacement")]
13    pub truck_placement: TruckPlacement,
14    /// Floating-point telemetry values.
15    #[serde(rename = "truckFloatValues")]
16    pub truck_float_values: TruckFloatValues,
17    /// Navigation speed limit in km/h.
18    #[serde(rename = "navigationSpeedLimit")]
19    #[serde(default)]
20    pub navigation_speed_limit: Option<f64>,
21    /// Distance to lead vehicle in meters, if available.
22    #[serde(rename = "distanceToLeadVehicle")]
23    #[serde(default)]
24    pub lead_vehicle_distance_m: Option<f32>,
25    /// Local longitudinal acceleration proxy (m/s²), if available.
26    #[serde(rename = "localAccelerationLongitudinal")]
27    #[serde(default)]
28    pub local_acceleration_longitudinal: Option<f32>,
29}
30
31/// Position and heading of the truck in world coordinates.
32#[derive(Debug, Clone, Deserialize)]
33pub struct TruckPlacement {
34    /// X coordinate in ETS2 world space.
35    pub x: f64,
36    /// Y coordinate in ETS2 world space (height).
37    pub y: f64,
38    /// Z coordinate in ETS2 world space.
39    pub z: f64,
40    /// Heading angle in radians.
41    pub heading: f64,
42    /// Pitch angle in radians.
43    #[serde(default)]
44    pub pitch: f64,
45    /// Roll angle in radians.
46    #[serde(default)]
47    pub roll: f64,
48}
49
50/// Floating-point truck values (speed, fuel, etc.).
51#[derive(Debug, Clone, Deserialize)]
52pub struct TruckFloatValues {
53    /// Current speed in m/s (converted from km/h by the server).
54    #[serde(default)]
55    pub speed: f64,
56    /// Engine RPM.
57    #[serde(default)]
58    pub engine_rpm: f64,
59    /// Fuel level in liters.
60    #[serde(default)]
61    pub fuel: f64,
62    /// Odometer reading in km.
63    #[serde(default)]
64    pub odometer: f64,
65    /// Cruise control speed in km/h.
66    #[serde(default)]
67    pub cruise_control_speed: f64,
68}
69
70/// Fetch the current telemetry snapshot from the ETS2 telemetry server.
71///
72/// Returns `Err` if the server is unreachable or the response cannot be parsed.
73pub fn fetch_telemetry(server_url: &str) -> Result<TelemetryData, Box<dyn std::error::Error>> {
74    let body = ureq::get(server_url)
75        .call()
76        .map_err(|e| format!("HTTP request failed: {e}"))?
77        .into_body()
78        .read_to_string()?;
79    let data: TelemetryData = serde_json::from_str(&body)?;
80    Ok(data)
81}

87#[cfg(test)]
88mod tests {
89    use super::*;
90
91    #[test]
92    fn test_telemetry_json_deserialization() {
93        let json = r#"
94        {
95            "truckPlacement": {
96                "x": 12345.6,
97                "y": 78.9,
98                "z": -1234.5,
99                "heading": 1.57,
100                "pitch": 0.1,
101                "roll": -0.05
102            },
103            "truckFloatValues": {
104                "speed": 25.0,
105                "engine_rpm": 1500.0,
106                "fuel": 350.5,
107                "odometer": 123456.7,
108                "cruise_control_speed": 80.0
109            },
110            "navigationSpeedLimit": 90.0
111        }
112        "#;
113        let data: TelemetryData = serde_json::from_str(json).unwrap();
114        assert_eq!(data.truck_placement.x, 12345.6);
115        assert_eq!(data.truck_placement.y, 78.9);
116        assert_eq!(data.truck_placement.z, -1234.5);
117        assert_eq!(data.truck_placement.heading, 1.57);
118        assert_eq!(data.truck_placement.pitch, 0.1);
119        assert_eq!(data.truck_placement.roll, -0.05);
120        assert_eq!(data.truck_float_values.speed, 25.0);
121        assert_eq!(data.truck_float_values.engine_rpm, 1500.0);
122        assert_eq!(data.truck_float_values.fuel, 350.5);
123        assert_eq!(data.truck_float_values.odometer, 123456.7);
124        assert_eq!(data.truck_float_values.cruise_control_speed, 80.0);
125        assert_eq!(data.navigation_speed_limit, Some(90.0));
126        assert_eq!(data.lead_vehicle_distance_m, None);
127        assert_eq!(data.local_acceleration_longitudinal, None);
128    }
129}
